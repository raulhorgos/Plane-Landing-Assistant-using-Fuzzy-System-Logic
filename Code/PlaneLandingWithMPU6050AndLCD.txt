#include <Adafruit_MPU6050.h>
#include <Adafruit_Sensor.h>
#include <Wire.h>
#include <LiquidCrystal_I2C.h>
#include "FuzzyLib.h"

FuzzyLib fuzzy;
Adafruit_MPU6050 mpu;
LiquidCrystal_I2C lcd(0x27, 16, 2); // I2C address 0x27, 16x2 LCD

// Variables for MPU6050
float pitch = 0.0;

// Simulated Height
float height = 50.0;
float heightIncrement = 0.5; // Increment height by 0.5 each loop

void setup() {
    Serial.begin(9600); // For debugging

    // Initialize MPU6050
    if (!mpu.begin()) {
        Serial.println("Failed to find MPU6050 chip");
        while (1) {
            delay(10);
        }
    }
    Serial.println("MPU6050 Found!");

    // Configure MPU6050
    mpu.setAccelerometerRange(MPU6050_RANGE_8_G);
    mpu.setGyroRange(MPU6050_RANGE_500_DEG);
    mpu.setFilterBandwidth(MPU6050_BAND_21_HZ);
    delay(100);

    // Initialize LCD
    lcd.init();
    lcd.backlight();
    lcd.setCursor(0, 0);
    lcd.print("Adjust Angle:");

    // Define membership functions for Height (all triangular)
    float heightMFs[3][4] = {
        {50, 50, 80, 100},   // Low
        {80, 100, 120, 120}, // Medium
        {100, 120, 150, 150}  // High
    };
    int heightMFTypes[3] = {0, 0, 0}; // All triangular

    // Define membership functions for Angle (all triangular)
    float angleMFs[3][4] = {
        {0, 0, 12, 22.5},   // Low
        {12, 22.5, 33, 33}, // Medium
        {22.5, 33, 45, 45}   // High
    };
    int angleMFTypes[3] = {0, 0, 0}; // All triangular

    // Define membership functions for Adjustment Angle (all triangular)
    float adjAngleMFs[5][4] = {
        {0, 0, 10, 30},    // Very Low
        {10, 30, 50, 50},  // Low
        {30, 50, 70, 70},  // Medium
        {50, 70, 90, 90},  // High
        {70, 90, 100, 100} // Very High
    };
    int adjAngleMFTypes[5] = {0, 0, 0, 0, 0}; // All triangular

    // Set up the membership functions
    fuzzy.setInputMFs(heightMFs, heightMFTypes, angleMFs, angleMFTypes, 3, 3);
    fuzzy.setOutputMFs(adjAngleMFs, adjAngleMFTypes, 5);

    // Define rules
    int rules[9][3] = {
        {0, 0, 0}, // If Height is Low and Angle is Low then Very Low
        {1, 0, 1}, // If Height is Medium and Angle is Low then Low
        {2, 0, 3}, // If Height is High and Angle is Low then High
        {0, 1, 1}, // If Height is Low and Angle is Medium then Low
        {1, 1, 2}, // If Height is Medium and Angle is Medium then Medium
        {2, 1, 0}, // If Height is High and Angle is Medium then Very Low
        {0, 2, 4}, // If Height is Low and Angle is High then Very High
        {1, 2, 3}, // If Height is Medium and Angle is High then High
        {2, 2, 2}  // If Height is High and Angle is High then Medium
    };
    fuzzy.setRules(rules);
}

void loop() {
    // Read MPU6050 data
    sensors_event_t a, g, temp;
    mpu.getEvent(&a, &g, &temp);

    // Calculate pitch angle (in degrees) using accelerometer data
    pitch = atan2(-a.acceleration.x, sqrt(pow(a.acceleration.y, 2) + pow(a.acceleration.z, 2))) * 180 / PI;
    
    // Ensure pitch is within the expected range [0, 45]
    pitch = constrain(pitch, 0, 45);

    // Simulate Height by incrementing
    height += heightIncrement;
    if (height >= 150 || height <= 50) {
        heightIncrement = -heightIncrement; // Reverse direction at bounds
    }
    height = constrain(height, 50, 150);

    // Process the fuzzy logic
    fuzzy.fuzzify(height, pitch);
    fuzzy.evaluateRules();
    float adjustmentAngle = fuzzy.defuzzify();

    // Display the Adjustment Angle on the LCD
    lcd.setCursor(0, 1); // Second row
    lcd.print("                "); // Clear the line
    lcd.setCursor(0, 1);
    lcd.print(adjustmentAngle, 1); // Display with 1 decimal place
    lcd.print(" deg");

    // For debugging (optional)
    Serial.print("Height: "); Serial.print(height);
    Serial.print(" | Pitch (Angle): "); Serial.print(pitch);
    Serial.print(" | Adjustment Angle: "); Serial.println(adjustmentAngle);
    
    delay(500); // Update every 500ms
}
