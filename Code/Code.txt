#include <Adafruit_MPU6050.h>
#include <Adafruit_Sensor.h>
#include <Wire.h>
#include <LiquidCrystal_I2C.h>
#include "FuzzyLib.h"

FuzzyLib fuzzy;
Adafruit_MPU6050 mpu;
LiquidCrystal_I2C lcd(0x27, 16, 2); // I2C address 0x27, 16x2 LCD

// Variables for MPU6050
float pitch = 0.0;

// Simulated Height
float height = 50.0;
float heightIncrement = 0.5; // Increment height by 0.5 each loop

void setup() {
    Serial.begin(9600); // For debugging

    // Initialize MPU6050
    if (!mpu.begin()) {
        Serial.println("Failed to find MPU6050 chip");
        while (1) {
            delay(10);
        }
    }
    Serial.println("MPU6050 Found!");

    // Configure MPU6050
    mpu.setAccelerometerRange(MPU6050_RANGE_8_G);
    mpu.setGyroRange(MPU6050_RANGE_500_DEG);
    mpu.setFilterBandwidth(MPU6050_BAND_21_HZ);
    delay(100);

    // Initialize LCD
    lcd.init();
    lcd.backlight();
    lcd.setCursor(0, 0);
    lcd.print("Adjust Angle:");

    // Define membership functions for Height (all triangular)
    float heightMFs[3][4] = {
        {50, 50, 80, 100},   // Low
        {80, 100, 120, 120}, // Medium
        {100, 120, 150, 150}  // High
    };
    int heightMFTypes[3] = {0, 0, 0}; // All triangular

    // Define membership functions for Angle (all triangular)
    float angleMFs[3][4] = {
        {0, 0, 12, 22.5},   // Low
        {12, 22.5, 33, 33}, // Medium
        {22.5, 33, 45, 45}   // High
    };
    int angleMFTypes[3] = {0, 0, 0}; // All triangular

    // Define membership functions for Adjustment Angle (all triangular)
    float adjAngleMFs[5][4] = {
        {0, 0, 10, 30},    // Very Low
        {10, 30, 50, 50},  // Low
        {30, 50, 70, 70},  // Medium
        {50, 70, 90, 90},  // High
        {70, 90, 100, 100} // Very High
    };
    int adjAngleMFTypes[5] = {0, 0, 0, 0, 0}; // All triangular

    // Set up the membership functions
    fuzzy.setInputMFs(heightMFs, heightMFTypes, angleMFs, angleMFTypes, 3, 3);
    fuzzy.setOutputMFs(adjAngleMFs, adjAngleMFTypes, 5);

    // Define rules
    int rules[9][3] = {
        {0, 0, 0}, // If Height is Low and Angle is Low then Very Low
        {1, 0, 1}, // If Height is Medium and Angle is Low then Low
        {2, 0, 3}, // If Height is High and Angle is Low then High
        {0, 1, 1}, // If Height is Low and Angle is Medium then Low
        {1, 1, 2}, // If Height is Medium and Angle is Medium then Medium
        {2, 1, 0}, // If Height is High and Angle is Medium then Very Low
        {0, 2, 4}, // If Height is Low and Angle is High then Very High
        {1, 2, 3}, // If Height is Medium and Angle is High then High
        {2, 2, 2}  // If Height is High and Angle is High then Medium
    };
    fuzzy.setRules(rules);
}

void loop() {
    // Read MPU6050 data
    sensors_event_t a, g, temp;
    mpu.getEvent(&a, &g, &temp);

    // Calculate pitch angle (in degrees) using accelerometer data
    pitch = atan2(-a.acceleration.x, sqrt(pow(a.acceleration.y, 2) + pow(a.acceleration.z, 2))) * 180 / PI;
    
    // Ensure pitch is within the expected range [0, 45]
    pitch = constrain(pitch, 0, 45);

    // Simulate Height by incrementing
    height += heightIncrement;
    if (height >= 150 || height <= 50) {
        heightIncrement = -heightIncrement; // Reverse direction at bounds
    }
    height = constrain(height, 50, 150);

    // Process the fuzzy logic
    fuzzy.fuzzify(height, pitch);
    fuzzy.evaluateRules();
    float adjustmentAngle = fuzzy.defuzzify();

    // Display the Adjustment Angle on the LCD
    lcd.setCursor(0, 1); // Second row
    lcd.print("                "); // Clear the line
    lcd.setCursor(0, 1);
    lcd.print(adjustmentAngle, 1); // Display with 1 decimal place
    lcd.print(" deg");

    // For debugging (optional)
    Serial.print("Height: "); Serial.print(height);
    Serial.print(" | Pitch (Angle): "); Serial.print(pitch);
    Serial.print(" | Adjustment Angle: "); Serial.println(adjustmentAngle);
    
    delay(500); // Update every 500ms
}

// Pin definitions
const int heightPotPin = A0;  // Potentiometer for Height
const int anglePotPin = A1;   // Potentiometer for Angle
const int servoPin = 9;       // Servo motor for Adjustment Angle

void setup() {
    Serial.begin(9600);
    adjustmentServo.attach(servoPin);

    // Define membership functions for Height (all triangular)
    float heightMFs[3][4] = {
        {50, 50, 80, 100},   // Low
        {80, 100, 120, 120}, // Medium
        {100, 120, 150, 150}  // High
    };
    int heightMFTypes[3] = {0, 0, 0}; // All triangular

    // Define membership functions for Angle (all triangular)
    float angleMFs[3][4] = {
        {0, 0, 12, 22.5},   // Low
        {12, 22.5, 33, 33}, // Medium
        {22.5, 33, 45, 45}   // High
    };
    int angleMFTypes[3] = {0, 0, 0}; // All triangular

    // Define membership functions for Adjustment Angle (all triangular)
    float adjAngleMFs[5][4] = {
        {0, 0, 10, 30},    // Very Low
        {10, 30, 50, 50},  // Low
        {30, 50, 70, 70},  // Medium
        {50, 70, 90, 90},  // High
        {70, 90, 100, 100} // Very High
    };
    int adjAngleMFTypes[5] = {0, 0, 0, 0, 0}; // All triangular

    // Set up the membership functions
    fuzzy.setInputMFs(heightMFs, heightMFTypes, angleMFs, angleMFTypes, 3, 3);
    fuzzy.setOutputMFs(adjAngleMFs, adjAngleMFTypes, 5);

    // Define rules (from the previous setup)
    int rules[9][3] = {
        {0, 0, 0}, // If Height is Low and Angle is Low then Very Low
        {1, 0, 1}, // If Height is Medium and Angle is Low then Low
        {2, 0, 3}, // If Height is High and Angle is Low then High
        {0, 1, 1}, // If Height is Low and Angle is Medium then Low
        {1, 1, 2}, // If Height is Medium and Angle is Medium then Medium
        {2, 1, 0}, // If Height is High and Angle is Medium then Very Low
        {0, 2, 4}, // If Height is Low and Angle is High then Very High
        {1, 2, 3}, // If Height is Medium and Angle is High then High
        {2, 2, 2}  // If Height is High and Angle is High then Medium
    };
    fuzzy.setRules(rules);
}

void loop() {
    // Read the potentiometer values (0 to 1023)
    int heightRaw = analogRead(heightPotPin);
    int angleRaw = analogRead(anglePotPin);

    // Map the raw values to the respective ranges
    float height = map(heightRaw, 0, 1023, 50, 150);  // Height range: 50 to 150
    float angle = map(angleRaw, 0, 1023, 0, 45);      // Angle range: 0 to 45

    // Process the fuzzy logic steps
    fuzzy.fuzzify(height, angle);
    fuzzy.evaluateRules();
    float adjustmentAngle = fuzzy.defuzzify();

    // Map the adjustment angle (0 to 100) to servo range (0 to 180 degrees)
    int servoAngle = map(adjustmentAngle, 0, 100, 0, 180);
    adjustmentServo.write(servoAngle);

    // Output the result to Serial Monitor
    Serial.print("Height: "); Serial.print(height);
    Serial.print(" | Angle: "); Serial.print(angle);
    Serial.print(" | Adjustment Angle: "); Serial.print(adjustmentAngle);
    Serial.print(" | Servo Angle: "); Serial.println(servoAngle);
    delay(500); // Update every 500ms
}

#include <Arduino.h>  // For min() and max()
#include "FuzzyLib.h"

FuzzyLib::FuzzyLib() {
    numHeightMFs = 0;
    numAngleMFs = 0;
    numAdjAngleMFs = 0;
}

void FuzzyLib::setInputMFs(float heightMFs[][4], int heightMFTypes[], float angleMFs[][4], int angleMFTypes[], int numHeightMFs, int numAngleMFs) {
    this->numHeightMFs = numHeightMFs;
    this->numAngleMFs = numAngleMFs;
    for (int i = 0; i < numHeightMFs; i++) {
        for (int j = 0; j < 4; j++) {
            this->heightMFs[i][j] = heightMFs[i][j];
        }
        this->heightMFTypes[i] = heightMFTypes[i];
    }
    for (int i = 0; i < numAngleMFs; i++) {
        for (int j = 0; j < 4; j++) {
            this->angleMFs[i][j] = angleMFs[i][j];
        }
        this->angleMFTypes[i] = angleMFTypes[i];
    }
}

void FuzzyLib::setOutputMFs(float adjAngleMFs[][4], int adjAngleMFTypes[], int numAdjAngleMFs) {
    this->numAdjAngleMFs = numAdjAngleMFs;
    for (int i = 0; i < numAdjAngleMFs; i++) {
        for (int j = 0; j < 4; j++) {
            this->adjAngleMFs[i][j] = adjAngleMFs[i][j];
        }
        this->adjAngleMFTypes[i] = adjAngleMFTypes[i];
    }
}

void FuzzyLib::setRules(int rules[][3]) {
    for (int i = 0; i < MAX_RULES; i++) {
        for (int j = 0; j < 3; j++) {
            this->rules[i][j] = rules[i][j];
        }
    }
}

float FuzzyLib::triMf(float x, float a, float b, float c) {
    if (x <= a || x >= c) return 0.0;
    if (x == b) return 1.0;
    if (x < b) return (x - a) / (b - a);
    return (c - x) / (c - b);
}

float FuzzyLib::trapMf(float x, float a, float b, float c, float d) {
    if (x <= a || x >= d) return 0.0;
    if (x >= b && x <= c) return 1.0;
    if (x < b) return (x - a) / (b - a);
    return (d - x) / (d - c);
}

void FuzzyLib::fuzzify(float height, float angle) {
    for (int i = 0; i < numHeightMFs; i++) {
        if (heightMFTypes[i] == 0) { // Triangular
            heightDegrees[i] = triMf(height, heightMFs[i][0], heightMFs[i][1], heightMFs[i][2]);
        } else { // Trapezoidal
            heightDegrees[i] = trapMf(height, heightMFs[i][0], heightMFs[i][1], heightMFs[i][2], heightMFs[i][3]);
        }
    }
    for (int i = 0; i < numAngleMFs; i++) {
        if (angleMFTypes[i] == 0) { // Triangular
            angleDegrees[i] = triMf(angle, angleMFs[i][0], angleMFs[i][1], angleMFs[i][2]);
        } else { // Trapezoidal
            angleDegrees[i] = trapMf(angle, angleMFs[i][0], angleMFs[i][1], angleMFs[i][2], angleMFs[i][3]);
        }
    }
}

void FuzzyLib::evaluateRules() {
    for (int i = 0; i < MAX_RULES; i++) {
        int heightIdx = rules[i][0];
        int angleIdx = rules[i][1];
        if (heightIdx == -1 || angleIdx == -1) {
            ruleFiringDegrees[i] = 0.0;
            continue;
        }
        // Use min (AND) for rule firing degree
        ruleFiringDegrees[i] = min(heightDegrees[heightIdx], angleDegrees[angleIdx]);
    }
}

void FuzzyLib::aggregate(float* outputDegrees) {
    for (int i = 0; i < numAdjAngleMFs; i++) {
        outputDegrees[i] = 0.0;
    }
    for (int i = 0; i < MAX_RULES; i++) {
        int adjAngleIdx = rules[i][2];
        if (adjAngleIdx != -1 && ruleFiringDegrees[i] > 0) {
            outputDegrees[adjAngleIdx] = max(outputDegrees[adjAngleIdx], ruleFiringDegrees[i]);
        }
    }
}

    float numerator = 0.0;
    float denominator = 0.0;
    const int samples = 1000;
    float step = 100.0 / (samples - 1); // Adjustment Angle range [0, 100]
    
    for (int i = 0; i < samples; i++) {
        float x = i * step;
        float membership = 0.0;
        for (int j = 0; j < numAdjAngleMFs; j++) {
            float degree;
            if (adjAngleMFTypes[j] == 0) { // Triangular
                degree = min(outputDegrees[j], triMf(x, adjAngleMFs[j][0], adjAngleMFs[j][1], adjAngle[j]MFs[j][2]);
            } else { // Trapezoidal
                degree = min(outputDegrees[j], trapMf(x, adjAngleMFs[j][0], adjAngleMFs[j][1], adjAngleMFs[j][2], adjAngleMFs[j][3]));
            }
            membership = max(membership, degree);
        }
        numerator += x * membership;
        denominator += membership;
    }
    
    if (denominator == 0) return 0.0;
    return numerator / denominator;
